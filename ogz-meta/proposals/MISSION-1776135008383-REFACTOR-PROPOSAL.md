  recordTradeOutcome(tradeData) {
    if (!this.traiCore) {
      console.log('⚠️ [TRAI] Cannot record trade - TRAI Core not initialized');
      return;
    }

    const tradeId = tradeData.tradeId;
    if (!tradeId) {
      console.warn('⚠️ [TRAI] tradeData missing tradeId – cannot aggregate');
      return;
    }

    // Retrieve existing aggregation or initialise a new one
    const existing = this.pendingOutcomes.get(tradeId) || {
      ...tradeData,
      profitLoss: 0,
      profitLossPercent: 0,
      holdDuration: 0,
      remainingSize: tradeData.remainingSize || 0
    };

    // Accumulate partial P&L and duration
    existing.profitLoss += tradeData.profitLoss || 0;
    existing.profitLossPercent += tradeData.profitLossPercent || 0;
    existing.holdDuration = Math.max(existing.holdDuration, tradeData.holdDuration || 0);
    // Update remainingSize if provided (partial exit)
    if (typeof tradeData.remainingSize !== 'undefined') {
      existing.remainingSize = tradeData.remainingSize;
    }

    // Store back the aggregated record
    this.pendingOutcomes.set(tradeId, existing);

    // Only fire learning sample when the trade is fully closed
    if (existing.remainingSize === 0) {
      try {
        this.traiCore.recordTradeResult(existing);
        console.log(`📚 [TRAI] Recorded trade outcome: ${existing.profitLoss > 0 ? 'WIN' : 'LOSS'} (${existing.profitLossPercent.toFixed(2)}%)`);
      } catch (error) {
        console.error('❌ [TRAI] Error recording trade outcome:', error.message);
      }
      this.pendingOutcomes.delete(tradeId);
    }
  }
